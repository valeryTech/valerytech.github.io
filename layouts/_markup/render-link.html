{{- $url := urls.Parse .Destination -}} {{- $scheme := $url.Scheme -}} {{- /* PDF links open the inline reader; without JavaScript they open the file. */ -}} {{- $path := lower $url.Path -}} {{- /* Imported PDFs are encrypted under opaque names in /sources/ (see scripts/migrate/paths.py). */ -}} {{- $isLocked := and (eq $scheme "") (strings.HasPrefix $path "/sources/") (strings.HasSuffix $path ".bin") -}} {{- $isPdf := or $isLocked (and (in (slice "" "http" "https") $scheme) (strings.HasSuffix $path ".pdf")) -}} {{- if $isPdf }}{{ .Page.Store.Set "hasPdfReader" true }}{{ end -}} {{- .Text | safeHTML -}} {{- /* whitespace stripped here to avoid trailing newline in rendered result caused by file EOL */ -}}
