{{- /* Require editorial descriptions, while allowing navigation and utility pages. */ -}} {{- $policy := site.Params.seo.description -}} {{- $validate := not (or (in $policy.exemptKinds .Kind) (in $policy.exemptTypes .Type)) -}} {{- if and (eq .Kind "section") (not (.Description | strings.TrimSpace)) (not .Params.seo.description) (not .RawSummary) (not (.RawContent | strings.TrimSpace)) -}} {{- $validate = false -}} {{- end -}} {{- if findRE `(?i)(^|[,\s])(noindex|none)($|[,\s])` (default "" .Params.seo.robots) -}} {{- $validate = false -}} {{- end -}} {{- range $policy.lengthIgnores -}} {{- if findRE . $.TranslationKey -}} {{- $validate = false -}} {{- end -}} {{- end -}} {{- /* An explicit boolean can opt an editorial landing page in or a utility out. */ -}} {{- if and .Params.seo (isset .Params.seo "validatedescription") -}} {{- $validate = .Params.seo.validateDescription -}} {{- end -}} {{- return $validate -}}
